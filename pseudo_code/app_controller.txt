PS DU FICHIER : controller/app_controller.py
OBJECTIF :
- Le contrôleur reçoit des actions UI (clics boutons)
- Met à jour self.state
- Demande à l’UI d’afficher self.state via self._render()

------------------------------------------------------------
RAPPEL : champs de self.state utilisés ici
- expression_text : string (ce que l’utilisateur tape)
- display_text    : string (ce qu’on affiche)
- message_text    : string (texte message)
- message_type    : string ("error" / "info" / "")
- error_mode      : bool (si True => on est en erreur)
- after_result    : bool (si True => dernier affichage était un résultat)
- recall_history  : bool (flag utilisé ailleurs, ici on le remet souvent à False)
- last_result_value : float | None
- last_result_text  : string
- mode_scientifique : bool
------------------------------------------------------------


CLASSE AppController
============================================================

fonction __init__(initial_state, ui):
    1) self.state = initial_state
    2) self.ui = ui


"""
HELPERS (petites fonctions internes)
"""

fonction _render():
    # 1) afficher l’état dans l’interface
    ui.render(self.state)

    # 2) IMPORTANT : les widgets d’historique sont recréés à chaque render
    #    donc on rebinde les events de clic d’historique
    ui.bind_history_clicks(self)


fonction _set_error(code):
    # Met l’app en "mode erreur" + message adapté
    1) self.state.message_text = get_message(code)
    2) self.state.message_type = "error"
    3) self.state.error_mode = True


fonction _set_info(text):
    # Met un message d’info simple (sans mode erreur)
    1) self.state.message_text = text
    2) self.state.message_type = "info"
    3) self.state.error_mode = False


fonction _clear_message():
    # Supprime le message affiché
    1) self.state.message_text = ""
    2) self.state.message_type = ""


fonction _clear_entry_keep_history_memory():
    # Bouton C : on efface uniquement la saisie actuelle + messages
    # (on ne touche pas à l’historique ni à la mémoire)
    1) self.state.expression_text = ""
    2) self.state.display_text = ""
    3) appeler _clear_message()
    4) self.state.error_mode = False
    5) self.state.after_result = False
    6) self.state.recall_history = False


fonction _append_token(token) -> bool:
    # Ajoute un token dans expression_text, MAIS seulement si la validation OK
    1) (ok, err) = validate_button_action(self.state.expression_text, token)

    2) SI ok == False ALORS
           appeler _set_error(err OU "UNKNOWN_TOKEN")
           retourner False

    3) SINON (ok == True)
           self.state.expression_text = self.state.expression_text + token
           self.state.display_text = self.state.expression_text
           appeler _clear_message()
           self.state.after_result = False
           self.state.recall_history = False
           retourner True


fonction _append_chars_as_clicks(text) -> bool:
    # Sert pour MR : injecter un nombre comme si l’utilisateur cliquait bouton par bouton
    # Exemple "-12.5" => '-', '1', '2', '.', '5'

    1) POUR chaque caractère ch dans text FAIRE
           SI ch == "-" ALORS
               SI _append_token("-") == False => retourner False

           SINON SI ch est un chiffre (0..9) ALORS
               SI _append_token(ch) == False => retourner False

           SINON SI ch == "." ALORS
               SI _append_token(".") == False => retourner False

           SINON
               appeler _set_error("UNKNOWN_TOKEN")
               retourner False

    2) retourner True


"""
HANDLERS (1 fonction = 1 action/bouton)
"""

fonction handle_digit(digit):
    # Boutons 0..9

    1) SI self.state.error_mode == True ALORS
           appeler _clear_entry_keep_history_memory()
           # (pas de return : on continue et on ajoute le chiffre)

    2) SI self.state.after_result == True ALORS
           # règle : taper un chiffre après un résultat => nouvelle expression
           appeler _clear_entry_keep_history_memory()

    3) appeler _append_token(digit)
    4) appeler _render()


fonction handle_operator(op):
    # + - * / % ^

    1) SI self.state.error_mode == True ALORS
           appeler _render()
           retourner

    2) SI self.state.after_result == True ALORS
           # règle : on enchaîne à partir du dernier résultat
           SI self.state.last_result_text est vide ALORS
               appeler _set_error("NO_LAST_RESULT")
               appeler _render()
               retourner
           SINON
               self.state.expression_text = self.state.last_result_text
               self.state.display_text = self.state.expression_text
               self.state.after_result = False

    3) appeler _append_token(op)
    4) appeler _render()


fonction handle_decimal():
    # bouton "."
    1) SI self.state.error_mode == True ALORS
           appeler _render()
           retourner

    2) appeler _append_token(".")
    3) appeler _render()


fonction handle_parenthesis(par):
    # bouton "(" ou ")"
    1) SI self.state.error_mode == True ALORS
           appeler _render()
           retourner

    2) SI self.state.after_result == True ET par == "(" ALORS
           # règle : ouvrir une parenthèse après résultat => nouvelle expression
           appeler _clear_entry_keep_history_memory()

    3) appeler _append_token(par)
    4) appeler _render()


fonction handle_scientific_token(token):
    # tokens possibles (selon l’UI) :
    # - "sqrt(" "abs(" "inv(" ...
    # - "^" "%"
    # - "!" (postfix)
    # - "^2" "^3" (cas spécial)

    1) SI self.state.error_mode == True ALORS
           appeler _render()
           retourner

    2) SI token == "^2" ALORS
           ok1 = _append_token("^")
           SI ok1 == True ALORS
               ok2 = _append_token("2")   # (dans le code ok2 existe même si non utilisé)
           appeler _render()
           retourner

    3) SI token == "^3" ALORS
           ok1 = _append_token("^")
           SI ok1 == True ALORS
               ok2 = _append_token("3")
           appeler _render()
           retourner

    4) SINON (cas général)
           appeler _append_token(token)
           appeler _render()


fonction handle_backspace():
    # bouton ⌫

    1) SI self.state.error_mode == True ALORS
           appeler _clear_entry_keep_history_memory()
           appeler _render()
           retourner

    2) SI self.state.expression_text est vide ALORS
           appeler _render()
           retourner

    3) self.state.expression_text = self.state.expression_text sans le dernier caractère
    4) self.state.display_text = self.state.expression_text
    5) appeler _clear_message()
    6) appeler _render()


fonction handle_clear():
    # bouton C
    1) appeler _clear_entry_keep_history_memory()
    2) appeler _render()


fonction handle_equals():
    # bouton "="

    1) SI self.state.error_mode == True ALORS
           appeler _render()
           retourner

    2) expr = self.state.expression_text

    3) (ok, err) = validate_expression_final(expr)
       SI ok == False ALORS
           appeler _set_error(err OU "INCOMPLETE_EXPR")
           appeler _render()
           retourner

    4) (success, value, engine_err) = evaluate_expression(expr)
       SI success == False OU value est None ALORS
           appeler _set_error(engine_err OU "EVAL_ERROR")
           appeler _render()
           retourner

    5) result_text = format_result(value)

    6) self.state.last_result_value = value
       self.state.last_result_text = result_text

    7) self.state = history_add(self.state, expr, result_text, value)

    8) self.state.expression_text = result_text
       self.state.display_text = result_text
       self.state.after_result = True
       self.state.error_mode = False
       appeler _clear_message()

    9) appeler _render()


fonction handle_history_click(index):
    1) (self.state, _) = history_select(self.state, index)
    2) appeler _render()


fonction handle_history_clear():
    1) self.state = history_clear(self.state)
    2) appeler _render()


fonction handle_memory_action(action):
    # actions : "MC" "MR" "M+" "M-"

    1) SI action == "MC" ALORS
           self.state = memory_clear(self.state)
           appeler _render()
           retourner

    2) SI action == "MR" ALORS
           (self.state, injected) = memory_recall(self.state)

           SI self.state.after_result == True ALORS
               appeler _clear_entry_keep_history_memory()

           ok = _append_chars_as_clicks(injected)
           SI ok == False ALORS
               # rien à faire : l’erreur a déjà été mise dans self.state
               pass

           appeler _render()
           retourner

    3) SI action == "M+" ALORS
           self.state = memory_add(self.state)
           appeler _render()
           retourner

    4) SI action == "M-" ALORS
           self.state = memory_sub(self.state)
           appeler _render()
           retourner

    5) SINON (action inconnue)
           appeler _set_error("UNKNOWN_TOKEN")
           appeler _render()


fonction toggle_scientific_mode():
    1) self.state.mode_scientifique = NON self.state.mode_scientifique
    2) appeler _render()


fonction handle_reset_all():
    # bouton AC (reset complet)
    1) self.state = create_initial_state()
    2) appeler _render()
