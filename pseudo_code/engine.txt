PS DU FICHIER : domain/engine.py

BUT :
- Calculer une expression écrite en texte SANS utiliser eval() et SANS utiliser math.
- On fait une méthode “classique” expliquable :
  1) tokenize() : transforme le texte en liste de tokens (nombres + symboles)
  2) to_rpn()   : convertit en notation polonaise inversée (RPN) avec shunting-yard
  3) eval_rpn() : évalue la RPN avec une pile

TOKENS possibles :
- nombres : float
- opérateurs : "+", "-", "*", "/", "%", "^"
- parenthèses : "(", ")"
- moins unaire : "u-"   (ex: "-5" ou "(-3)")
- fonctions : "sqrt", "abs", "inv"
- factorielle postfix : "!"

Les fonctions scientifiques sont calculées dans domain/scientific.py :
- pow_int, sqrt_newton, factorial_int, abs_value, inv_value, mod_int


"""
1) evaluate_expression(expr)
"""

fonction evaluate_expression(expr) -> (success, value, error_code):
    # Objectif : fonction “top niveau” utilisée par le controller

    1) tokens = tokenize(expr)
       SI tokens == None ALORS
           retourner (False, None, "TOKEN_ERROR")

    2) rpn = to_rpn(tokens)
       SI rpn == None ALORS
           retourner (False, None, "PARSE_ERROR")

    3) (ok, value, err) = eval_rpn(rpn)
       SI ok == False ALORS
           retourner (False, None, err)

    4) retourner (True, value, None)


"""
2) tokenize(expr)
"""

fonction tokenize(expr) -> liste de tokens OU None:
    # Objectif : lire le texte caractère par caractère et reconnaître :
    # - nombres (float)
    # - fonctions "sqrt(" "abs(" "inv("
    # - opérateurs et parenthèses
    # - factorielle "!"
    # - moins unaire -> token spécial "u-"

    1) tokens = []
    2) i = 0
       n = len(expr)

    3) fonction interne last_is_operator_or_open_paren() -> bool:
           # sert uniquement à reconnaître le - unaire
           SI tokens est vide => True
           last = dernier token
           retourner True si last est une string dans {"+","-","*","/","%","^","u-","("}
           sinon False

    4) TANT QUE i < n :
           c = expr[i]

           a) SI c est un espace :
                 i += 1
                 continuer

           b) SI expr commence par "sqrt(" à la position i :
                 tokens.append("sqrt")
                 tokens.append("(")
                 i += longueur("sqrt(")
                 continuer

              SI expr commence par "abs(" à la position i :
                 tokens.append("abs")
                 tokens.append("(")
                 i += longueur("abs(")
                 continuer

              SI expr commence par "inv(" à la position i :
                 tokens.append("inv")
                 tokens.append("(")
                 i += longueur("inv(")
                 continuer

           c) SI c est un chiffre OU c == "." :
                 # on lit un nombre complet (avec max 1 point)
                 j = i
                 dot_count = 0
                 TANT QUE j < n ET (expr[j] est chiffre OU expr[j] == "."):
                      SI expr[j] == "." :
                          dot_count += 1
                          SI dot_count > 1 => retourner None
                      j += 1

                 part = expr[i:j]
                 SI part == "." => retourner None

                 essayer tokens.append(float(part))
                 SI erreur => retourner None

                 i = j
                 continuer

           d) SI c est "(" OU ")" :
                 tokens.append(c)
                 i += 1
                 continuer

           e) SI c == "!" :
                 tokens.append("!")
                 i += 1
                 continuer

           f) SI c est un opérateur dans {"+","-","*","/","%","^"} :
                 SI c == "-" ET last_is_operator_or_open_paren() == True :
                     # cas - unaire (ex: "-5", "(-3", "2*-4")
                     tokens.append("u-")
                 SINON :
                     tokens.append(c)

                 i += 1
                 continuer

           g) SINON :
                 # caractère inconnu
                 retourner None

    5) retourner tokens


"""
3) precedence(op) et helpers
"""

fonction precedence(op) -> int:
    # priorité la plus haute = évaluée en premier
    SI op == "!" => 5
    SI op dans {"u-", "sqrt", "abs", "inv"} => 4
    SI op == "^" => 3
    SI op dans {"*", "/", "%"} => 2
    SI op dans {"+", "-"} => 1
    sinon 0

fonction is_right_associative(op) -> bool:
    # associativité droite pour ^ et u-
    retourner True si op dans {"^","u-"} sinon False

fonction is_operator(tok) -> bool:
    retourner True si tok est une string dans {"+","-","*","/","%","^","u-","!"}

fonction is_function(tok) -> bool:
    retourner True si tok est une string dans {"sqrt","abs","inv"}


"""
4) to_rpn(tokens)
"""

fonction to_rpn(tokens) -> liste RPN OU None:
    # Objectif : convertir tokens vers une forme facile à calculer (RPN)
    # RPN = on n’a plus besoin de parenthèses, on lit de gauche à droite avec pile

    1) output = []
    2) stack = []   # pile de strings (opérateurs, parenthèses, fonctions)

    3) POUR chaque tok dans tokens :

         a) SI tok est un float :
               output.append(tok)
               continuer

         b) SI tok est une fonction (sqrt/abs/inv) :
               stack.append(tok)
               continuer

         c) SI tok est un opérateur (+-*/%^u-!) :
               # On dépile tant que le haut de pile est prioritaire
               TANT QUE stack n’est pas vide :
                    top = stack[-1]
                    SI top == "(" :
                        arrêter

                    SI top est une fonction OU un opérateur :
                        SI precedence(top) > precedence(tok) :
                            output.append(stack.pop())
                            continuer

                        SI precedence(top) == precedence(tok) ET is_right_associative(tok) == False :
                            output.append(stack.pop())
                            continuer

                    arrêter

               stack.append(tok)
               continuer

         d) SI tok == "(" :
               stack.append("(")
               continuer

         e) SI tok == ")" :
               # On dépile jusqu'à trouver "("
               found_open = False
               TANT QUE stack n’est pas vide :
                    top = stack.pop()
                    SI top == "(" :
                        found_open = True
                        arrêter
                    output.append(top)

               SI found_open == False :
                    retourner None   # parenthèses incohérentes

               # si juste avant il y avait une fonction, on la met aussi dans output
               SI stack non vide ET stack[-1] est une fonction :
                    output.append(stack.pop())

               continuer

         f) SINON :
               retourner None

    4) Fin de lecture : vider la pile
       TANT QUE stack non vide :
           top = stack.pop()
           SI top est "(" OU ")" :
               retourner None
           output.append(top)

    5) retourner output


"""
5) eval_rpn(rpn)
"""

fonction eval_rpn(rpn) -> (ok, value, error_code):
    # Objectif : calculer la RPN avec une pile de nombres
    # Règle : quand on voit un opérateur, on prend les valeurs sur la pile

    1) stack = []  # pile de float

    2) POUR chaque tok dans rpn :

         a) SI tok est un float :
               stack.append(tok)
               continuer

         b) SI tok == "u-" :
               SI stack a moins de 1 valeur :
                   retourner (False, None, "BAD_UNARY_MINUS")
               a = stack.pop()
               stack.append(-a)
               continuer

         c) SI tok est un opérateur binaire dans {"+","-","*","/","%","^"} :
               SI stack a moins de 2 valeurs :
                   retourner (False, None, "BAD_BINARY_OP")

               b = stack.pop()
               a = stack.pop()

               SI tok == "+" : stack.append(a + b)
               SI tok == "-" : stack.append(a - b)
               SI tok == "*" : stack.append(a * b)

               SI tok == "/" :
                   SI b == 0 : retourner (False, None, "DIV_ZERO")
                   stack.append(a / b)

               SI tok == "%" :
                   res = mod_int(a, b)
                   SI res == None : retourner (False, None, "MOD_ERROR")
                   stack.append(res)

               SI tok == "^" :
                   res = pow_int(a, b)
                   SI res == None :
                       SI a == 0 ET b < 0 :
                           retourner (False, None, "POW_ZERO_NEG")
                       retourner (False, None, "POW_EXP_NOT_INT")
                   stack.append(res)

               continuer

         d) SI tok == "!" :
               SI stack a moins de 1 valeur :
                   retourner (False, None, "FACTORIAL_NO_ARG")
               a = stack.pop()
               res = factorial_int(a)
               SI res == None :
                   retourner (False, None, "FACTORIAL_NOT_INT")
               stack.append(res)
               continuer

         e) SI tok == "sqrt" :
               SI stack a moins de 1 valeur :
                   retourner (False, None, "SQRT_NO_ARG")
               a = stack.pop()
               res = sqrt_newton(a)
               SI res == None :
                   retourner (False, None, "SQRT_NEGATIVE")
               stack.append(res)
               continuer

         f) SI tok == "abs" :
               SI stack a moins de 1 valeur :
                   retourner (False, None, "ABS_NO_ARG")
               a = stack.pop()
               stack.append(abs_value(a))
               continuer

         g) SI tok == "inv" :
               SI stack a moins de 1 valeur :
                   retourner (False, None, "INV_NO_ARG")
               a = stack.pop()
               res = inv_value(a)
               SI res == None :
                   retourner (False, None, "INV_ZERO")
               stack.append(res)
               continuer

         h) SINON :
               retourner (False, None, "EVAL_ERROR")

    3) À la fin, il doit rester EXACTEMENT 1 valeur
       SI len(stack) != 1 :
           retourner (False, None, "EVAL_ERROR")

    4) retourner (True, stack[0], None)
