"""
Dictionnaire pour ERROR_MESSAGES
"""

ERROR_MESSAGES = {
    "EMPTY_EXPR": "Entrez une expression avant de calculer.",
    "MAX_LEN": "Expression trop longue.",
    ...
    "DIV_ZERO": "Division par zéro impossible.",
    ...
}

IDEe :
- la clé = code erreur (string)
- la valeur = message utilisateur (string)



fonction get_message(code, context=None) -> string:
    # Objectif :
    # - retrouver le message correspondant à un code
    # - si le code n’existe pas, retourner "Erreur inconnue."
    # - si context existe : remplacer les {variables} dans le message

    1) msg = ERROR_MESSAGES.get(code, "Erreur inconnue.")
       # .get() : si code est trouvé -> renvoie le message
       # sinon -> renvoie directement "Erreur inconnue."

    2) SI context n’est pas None ALORS
           POUR chaque (k, v) dans context.items():
               # On remplace les placeholders du style "{k}" par la valeur v
               msg = msg.replace("{" + str(k) + "}", str(v))

    3) retourner msg
