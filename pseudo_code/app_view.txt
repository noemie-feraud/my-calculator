fonction __init__():
    # Objectif : créer toute l’interface une seule fois

    1) Configurer CustomTkinter
       - ctk.set_appearance_mode("System")
       - ctk.set_default_color_theme("blue")
       - scaling widgets + fenêtre (UI plus grande)

    2) Créer la fenêtre principale
       - self.window = CTk()
       - titre + taille

    3) Créer la structure principale (2 colonnes)
       - self.left_frame  : zone calculatrice
       - self.right_frame : zone historique
       - placer en grid
       - configurer les weights (left plus large que right)

    4) Créer la zone d’affichage (NON éditable)
       - self.display_label (Label aligné à droite)
       - self.message_label (Label pour erreurs/infos)

    5) Créer la zone “clavier”
       - self.keyboard_frame
       - dedans : 2 frames côte à côte
           a) self.standard_frame (boutons classiques)
           b) self.scientific_frame (boutons scientifiques + mémoire)

    6) Créer la partie historique (à droite)
       - titre “Historique”
       - self.history_scroll (ScrollableFrame)
       - frame avec 2 boutons :
           - self.btn_clear_history
           - self.btn_reset_all

    7) Préparer des conteneurs pour stocker les boutons
       (ça simplifie le binding ensuite)
       - self.btns_digit : dict "0".."9" -> bouton
       - self.btns_op    : dict "+-*/" -> bouton
       - self.btn_decimal, self.btn_open_par, self.btn_close_par, etc.
       - self.btns_sci : dict token -> bouton scientifique
       - self.btns_mem : dict action -> bouton mémoire
       - self._history_row_buttons : liste des boutons créés pour l’historique

    8) Construire réellement les boutons
       - appeler self._build_standard_buttons()
       - appeler self._build_scientific_buttons()


fonction mainloop():

fonction _build_standard_buttons():
    # Objectif : créer les boutons "classiques" dans standard_frame

    1) Définir la grille (liste de lignes)
       [
         ["7","8","9","/"],
         ["4","5","6","*"],
         ["1","2","3","-"],
         ["0",".","=","+"],
         ["(",")","C","⌫"]
       ]

    2) Pour chaque case (r, c) de la grille :
       - créer un bouton CTkButton(text=label)
       - le placer en grid
       - selon le label, le ranger au bon endroit :
           SI label est un chiffre -> self.btns_digit[label] = btn
           SI label est "+-*/"     -> self.btns_op[label] = btn
           SI label == "."         -> self.btn_decimal = btn
           SI label == "("         -> self.btn_open_par = btn
           SI label == ")"         -> self.btn_close_par = btn
           SI label == "C"         -> self.btn_clear = btn
           SI label == "⌫"         -> self.btn_backspace = btn
           SI label == "="         -> self.btn_equals = btn
              + mise en forme spéciale (couleur)

    3) Créer le bouton "SCI" (toggle)
       - self.btn_sci_toggle
       - placé en dessous (row=5, columnspan=4)

"""
BINDING : CONNECTER LES BOUTONS AU CONTROLLER
"""

fonction bind_buttons(controller):
    # Objectif : relier chaque bouton à une fonction du controller
    # Ici on ne calcule rien : on fait juste "au clic -> controller...."

    1) Pour chaque chiffre (digit, btn) dans self.btns_digit :
         au clic -> controller.handle_digit(digit)

    2) Pour chaque opérateur (op, btn) dans self.btns_op :
         au clic -> controller.handle_operator(op)

    3) Boutons simples :
         "."  -> controller.handle_decimal()
         "("  -> controller.handle_parenthesis("(")
         ")"  -> controller.handle_parenthesis(")")
         "⌫"  -> controller.handle_backspace()
         "C"  -> controller.handle_clear()
         "="  -> controller.handle_equals()

    4) Toggle SCI :
         "SCI" -> controller.toggle_scientific_mode()

    5) Scientifique :
         pour chaque (token, btn) dans self.btns_sci :
             au clic -> controller.handle_scientific_token(token)

    6) Mémoire :
         pour chaque (action, btn) dans self.btns_mem :
             au clic -> controller.handle_memory_action(action)

    7) Actions historique (boutons à droite) :
         "Effacer historique" -> controller.handle_history_clear()
         "Réinitialiser (AC)" -> controller.handle_reset_all()


fonction bind_history_clicks(controller):
    # Objectif : brancher les clics sur les lignes d’historique
    # Important : on doit le refaire après chaque render
    # car les boutons d’historique sont détruits puis recréés

    pour i, btn dans enumerate(self._history_row_buttons):
        au clic -> controller.handle_history_click(i)


"""
RENDER : AFFICHER L’ETAT
"""

fonction render(state):
    # Objectif : mettre l’UI à jour avec state

    1) Mettre à jour le display :
         display_label.text = state.display_text

    2) Mettre à jour le message :
         SI state.message_type == "error":
             message_label.text = state.message_text
             message_label.couleur = rouge
         SINON SI state.message_type == "info":
             message_label.text = state.message_text
             message_label.couleur = vert
         SINON:
             message_label.text = ""
             message_label.couleur = gris

    3) Afficher/masquer le clavier scientifique :
         SI state.mode_scientifique == True:
             scientific_frame.grid()       # visible
         SINON:
             scientific_frame.grid_remove() # caché

    4) Mettre à jour l’historique :
         appeler _render_history(state.history_items)


fonction _render_history(items):
    # Objectif : reconstruire entièrement la liste d’historique à droite
    # (simple : on détruit tout puis on recrée)

    1) Détruire les anciens boutons :
         pour chaque btn dans self._history_row_buttons:
             btn.destroy()
         self._history_row_buttons = []

    2) Recréer un bouton par item :
         pour chaque item dans items:
             texte = "item.expr = item.result_text"
             btn = CTkButton(history_scroll, text=texte, anchor="w")
             btn.pack(...)
             ajouter btn dans self._history_row_buttons
